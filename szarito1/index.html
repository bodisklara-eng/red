def insert_at(rope,index,count,new_items):
    for iter in range(count):
        if all(rope): raise OverflowError('ESET4 teli')
        item=new_items[iter]
        if rope[index] is None:
            rope[index]=item # ESET3
        else:
            empty=next((j for j in range(index,len(rope)) if rope[j] is None),-1)
            if empty==-1: raise OverflowError('nincs üres jobbra')
            for k in range(empty,index,-1):
                rope[k]=rope[k-1] # jobbra tolás ESET1
            rope[index]=item
        index=min(index+1,len(rope)-1)

def remove_at(rope,index,count):
    for i in range(count):
        at=index+i
        if at < len(rope): rope[at]=None # ESET5 lyuk marad
