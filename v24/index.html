🐍 Python Animációs Kód (Matplotlib)

import matplotlib.pyplot as plt

sugar_ero = [80, 60, 40, 100, 70, 30, 90, 50, 60, 80, 80]
gyakorisag = {}
plt.ion()
fig, ax = plt.subplots()

for i, ertek in enumerate(sugar_ero):
    gyakorisag[ertek] = gyakorisag.get(ertek, 0) + 1
    ax.clear()
    ax.bar([str(k) for k in gyakorisag.keys()], gyakorisag.values(), color='#38bdf8')
    ax.set_title(f"Lépés {i+1}: Érték = {ertek}% | Helyzetvektor: {list(gyakorisag.values())}")
    plt.draw()
    plt.pause(0.6)

plt.ioff()
plt.show()
