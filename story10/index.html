Kapacitás: N = 20
Foglalt elemek: 10 db
Üres lyukak: 10 db
Kiválasztott index: Nincs
🧺 Beszúrás (INSERT_AT)
🧼 Levétel (REMOVE_AT)
ESET 1: Foglalt helyre beszúrás O(k)
Jobbra tolás elvégzése az első üres slotig (k = eltolások száma).
ESET 2/6: Belső lyuk kitöltése O(1)
Kijelölt vagy balról legközelebbi üres slot kitöltése eltolás nélkül.
ESET 3: Beszúrás üres slotra O(1)
Közvetlen értékadás, elemek mozgatása nem szükséges.
ESET 4: Overflow (Kötél megtelt) O(N)
Nincs üres hely jobbra, a beszúrási művelet meghiúsul.
ESET 5: Elem törlése (Levétel) O(1)
A slot értéke `null` lesz, automatikus tömörítés nélkül (lyuk marad).
ESET 7: Kötegelt beszúrás O(m * k)
Több ruha egymás utáni iteratív felakasztása és eltolása.
ESET 8: Defragmentálás (Compress) O(N)
Összes elem balra tömörítése, lyukak megszüntetése a tömb végén.

// JavaScript 1D Tömb Műveletek
function findEmptyRight(rope, index) {
    for (let i = index; i < rope.length; i++) {
        if (rope[i] === null) return i;
    }
    return -1;
}

function insertAt(rope, index, newCloth) {
    if (rope[index] === null) {
        rope[index] = newCloth; // ESET 3: O(1)
        return;
    }
    let emptyIdx = findEmptyRight(rope, index);
    if (emptyIdx === -1) throw new Error("OVERFLOW"); // ESET 4
    
    // ESET 1: Jobbra tolási ciklus O(k)
    for (let k = emptyIdx; k > index; k--) {
        rope[k] = rope[k - 1];
    }
    rope[index] = newCloth;
}

function removeAt(rope, index, count = 1) {
    // ESET 5: O(1) törlés (null értékadás)
    for (let i = 0; i < count && (index + i) < rope.length; i++) {
        rope[index + i] = null;
    }
}

function compress(rope) {
    // ESET 8: O(N) Defragmentálás balra
    let filled = rope.filter(item => item !== null);
    while (filled.length < rope.length) filled.push(null);
    return filled;
}
